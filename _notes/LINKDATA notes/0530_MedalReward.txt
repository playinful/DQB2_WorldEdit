# MINI MEDAL REWARDS #

0x00 = ushort(?) 0x0100 (256) (amount of entries in file)
0x02..0x3F = Empty
0x40..0x43F = uint[] Integers from 0-255
0x440..0103F = Reward Data[] (0xC bytes each) (256 count)
    +0x00 = ushort Item/Recipe ID
    +0x02..+0x03 = ?? (always 0)
    +0x04 = byte(?) Item or Recipe (1 = Item, 2 = Recipe)
    +0x05 = ?? (always 0)
    +0x06 = ?? (always 1)
    +0x07 = byte(?) Mini Medal Requirement
    +0x08..+0x0B = ?? (always 0)
